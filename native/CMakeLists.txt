cmake_minimum_required(VERSION 3.21)
project(BrowserNative LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "This native target currently supports macOS only")
endif()
set(PROJECT_ARCH "arm64" CACHE STRING "CEF architecture")
set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(USE_SANDBOX ON CACHE BOOL "CEF sandbox must remain enabled" FORCE)
if(NOT CEF_ROOT)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cef-path.txt" CEF_ROOT)
  string(STRIP "${CEF_ROOT}" CEF_ROOT)
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_executable(BrowserHelper Bridge/Helper.cc)
SET_EXECUTABLE_TARGET_PROPERTIES(BrowserHelper)
target_link_libraries(BrowserHelper PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
add_executable(BrowserCdmHelper Bridge/CdmHelper.cc)
SET_EXECUTABLE_TARGET_PROPERTIES(BrowserCdmHelper)
target_link_libraries(BrowserCdmHelper PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
# CEF 156's CefScopedSandboxContext loads libcef_sandbox.dylib from the
# embedded framework. It intentionally does not link a libcef_sandbox.a.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Bridge/BrowserBridge.mm")
  add_library(BrowserBridge STATIC Bridge/BrowserBridge.mm)
  SET_LIBRARY_TARGET_PROPERTIES(BrowserBridge)
  target_compile_options(BrowserBridge PRIVATE -fobjc-arc)
  target_include_directories(BrowserBridge PRIVATE "${CEF_ROOT}" Bridge)
  target_link_libraries(BrowserBridge PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
endif()
